  - a suite that draws Mermaid diagrams (class="mermaid") keeps the module

  The result is wrapped in 
: a suite is
  mostly code, so it spans the text and the margin columns

{{- $path := "" -}}
{{- $shift := 2 -}}
{{- if .IsNamedParams -}}
  {{- $path = .Get "path" | default "" -}}
  {{- with .Get "shift" }}{{ $shift = int . }}{{ end -}}
{{- else -}}
  {{- $path = .Get 0 | default "" -}}
{{- end -}}
{{- if not $path -}}
  {{- errorf "include shortcode: a file is required, e.g. {{< include \"test-suites/testsuite-x.html\" >}} (%s)" .Position -}}
{{- end -}}
{{- if not (fileExists $path) -}}
  {{- errorf "include shortcode: %q not found; run `make import-test-suites` in the hugo directory (%s)" $path .Position -}}
{{- end -}}
{{- $doc := readFile $path -}}

{{- /* The body (a fragment without  is taken whole) and the head. */ -}}
{{- $head := "" -}}
{{- with findRESubmatch `(?is)]*>(.*?)` $doc 1 }}{{ $head = index (index . 0) 1 }}{{ end -}}
{{- $html := $doc -}}
{{- with findRESubmatch `(?is)]*>(.*)` $doc 1 }}{{ $html = index (index . 0) 1 }}{{ end -}}
{{- $html = replaceRE `(?is)||]*>` "" $html -}}

{{- /* Sections are the theme's: the suite's become divs. */ -}}
{{- $html = replaceRE `(?i)` `
` $html -}}

{{- /* Unique ids. Two suites of one post may test functions of the same
       name (test/return in two category suites): the first suite on the
       page keeps the id, a later one gets "-2" ("-3"…), with the links to
       it. Each id remembers the suite that owns it (page Store, which a
       server rebuild does not reset), so re-rendering the page gives the
       same ids. Within one suite the ids are unique, and code holds no
       literal `id="` (its quotes are escaped). */ -}}
{{- $owners := .Page.Store.Get "includeIds" | default dict -}}
{{- range findRESubmatch ` id="([^"]+)"` $html -1 -}}
  {{- $id := index . 1 -}}
  {{- $owner := index $owners $id -}}
  {{- if and $owner (ne $owner $path) -}}
    {{- $new := "" -}}
    {{- range $n := seq 2 999 -}}
      {{- $cand := printf "%s-%d" $id $n -}}
      {{- $o := index $owners $cand -}}
      {{- if or (not $o) (eq $o $path) }}{{ $new = $cand }}{{ break }}{{ end -}}
    {{- end -}}
    {{- $html = replace $html (printf ` id="%s"` $id) (printf ` id="%s"` $new) -}}
    {{- $html = replace $html (printf ` href="#%s"` $id) (printf ` href="#%s"` $new) -}}
    {{- $.Page.Store.SetInMap "includeIds" $new $path -}}
  {{- else if not $owner -}}
    {{- $.Page.Store.SetInMap "includeIds" $id $path -}}
  {{- end -}}
{{- end -}}

{{- /* Headings down $shift levels, deepest first so that no heading moves
       twice; h6 is the floor. */ -}}
{{- if gt $shift 0 -}}
  {{- range seq 5 -1 1 -}}
    {{- $to := math.Min 6 (add . $shift) | int -}}
    {{- $html = replaceRE (printf `(?i)<(/?)h%d([\s>])` .) (printf `<${1}h%d${2}` $to) $html -}}
  {{- end -}}
{{- end -}}
