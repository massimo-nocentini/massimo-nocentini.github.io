(module
  (aux continuation classic)
  *
  (import
    scheme
    (chicken base)
    (chicken continuation)
    (only (chicken fixnum) fx+ fx- fx<)
    (only srfi-1 last)
    (aux base)
    (aux continuation))
  (define (continuation-rewind! winds n)
    (cond ((eq? ##sys#dynamic-winds winds) (void))
          ((fx< n 0)
           (continuation-rewind! (cdr winds) (fx+ n 1))
           ((car (car winds)))
           (set! ##sys#dynamic-winds winds))
          (else
           (let1 (after (cdr (car ##sys#dynamic-winds)))
                 (set! ##sys#dynamic-winds (cdr ##sys#dynamic-winds))
                 (after)
                 (continuation-rewind! winds (fx- n 1))))))
  (define continuation->λ
    (λ (raw)
       (##sys#check-structure raw 'continuation 'continuation->λ)
       (let1 (winds (##sys#slot raw 2))
             (λ args
                (unless
                  (eq? ##sys#dynamic-winds winds)
                  (continuation-rewind!
                    winds
                    (fx- (length ##sys#dynamic-winds) (length winds))))
                (apply continuation-return raw args)))))
  (define-syntax-rule
    (letcc/raw (raw hop) body ...)
    (continuation-capture
      (λ (raw) (let1 (hop (continuation->λ raw)) body ...))))
  (define (apply/cc f . args)
    (letcc k (apply f (append (butlast args) (last args) (list k)))))
  (define-syntax
    escapecc
    (syntax-rules
      (else =>)
      ((escapecc (out α) (else β)) (letcc out α β))
      ((escapecc (out α) (=> f)) (letcc out (let1 (result α) (f result))))))
  (define-syntax
    trycc/named
    (syntax-rules
      (else =>)
      ((trycc/named (skip α) ... (else β))
       (letcc success (letcc skip (success α)) ... β))
      ((trycc/named (skip α) ... (else => f))
       (letcc success
              (let1 (outputs (list (letcc skip (success α)) ...))
                    (f outputs))))))
  (define-syntax-rule (set/cc! v) (set! v (letcc/raw (raw hop) hop)))
  (define-syntax
    cond/cc
    (syntax-rules
      (else)
      ((cond/cc (pred? handling) ... (else recv))
       (letcc/raw
         (cont hop)
         (cond ((continuation? cont) (recv hop))
               ((pred? cont) (handling cont))
               ...
               (else (error "Contract violation")))))))
  (define current-continuation/cont
    (τ (continuation-capture (λ (cont) ((continuation->λ cont) cont)))))
  (define current-continuation/λ
    (τ (continuation-capture (compose Φ continuation->λ))))
  (define-syntax-rule
    (letcc/escaper escaper body ...)
    ((letcc top
            (let1 (escaper (λ (proc) (λ args (top (τ (apply proc args))))))
                  (τ body ...))))))

(define (test/sat-solve _)

  (⊦= (list (list #f #f #t))
      (sat-solve (a b c) (and (implies a (not b)) (not a) c)))
