  (define (hand-off-time n)
    (receive
      (r w)
      (create-pipe)
      (let1 (t (run-cml
                 (τ (for-each
                      (λ (i) (spawn (τ (sync (io-evt r 'input)))))
                      (iota n))
                    (cml/sleep 0.02)
                    (let ((ch (make-channel))
                          (t0 (current-process-milliseconds)))
                      (spawn (τ (let loop ((i 0))
                                  (when (< i 20000) (send ch i) (loop (add1 i))))))
                      (let loop ((i 0))
                        (when (< i 20000) (recv ch) (loop (add1 i))))
                      (cml/shutdown (- (current-process-milliseconds) t0))))))
            (file-close r)
            (file-close w)
            t)))
  (let* ((none (max 50 (hand-off-time 0))) (idle (hand-off-time 2000)))
    (⊨ (< idle (* 4 none)))))
