Introduction

This section tests the ziplist monad, whose bind zips rather than nests: the i-th element of (>>= m g) is the i-th element of (g v), where v is the i-th element of m. A pure value is the circular list that repeats it, so it zips with lists of any length.

Implementation

(module
  (aux category ziplist)
  *
  (import scheme (chicken base) (only srfi-1 take every iota) (aux base))
  (define (list-tail/or-false lst i)
    (cond ((null? lst) #f)
          ((zero? i) lst)
          (else (list-tail/or-false (cdr lst) (sub1 i)))))
  (define (list-rho lst)
    (let meet ((t lst) (h lst))
      (cond ((or (null? h) (null? (cdr h))) #f)
            ((eq? (cdr t) (cddr h))
             (let* ((t (cdr t))
                    (start (let S ((a lst) (b t) (n 0))
                             (if (eq? a b) n (S (cdr a) (cdr b) (add1 n)))))
                    (period
                      (let P ((b (cdr t)) (n 1))
                        (if (eq? b t) n (P (cdr b) (add1 n))))))
               (cons start period)))
            (else (meet (cdr t) (cddr h))))))
  (define (>>=/circular m start period g)
    (let* ((images (list->vector (map g (take m (+ start period)))))
           (at (λ (i)
                  (vector-ref
                    images
                    (if (< i start) i (+ start (modulo (- i start) period))))))
           (rhos (map (λ (k) (list-rho (vector-ref images k)))
                      (iota period start)))
           (from (and (every pair? rhos) (apply max start (map car rhos))))
           (cycle (and from (apply lcm period (map cdr rhos))))
           (bound (and from (+ from cycle)))
           (d (let D ((i 0) (acc '()))
                (cond ((eqv? i bound) (reverse acc))
                      ((list-tail/or-false (at i) i)
                       =>
                       (λ (t) (D (add1 i) (cons (car t) acc))))
                      (else (reverse acc))))))
      (when (and bound (= (length d) bound))
            (set-cdr! (list-tail d (sub1 bound)) (list-tail d from)))
      d))
  (define (return x) (list/circular x))
  (define (>>= m g)
    (match/first
      m
      ((,l unquote ,m) (g l))
      (else (match/first
              (list-rho m)
              ((,start unquote period) (>>=/circular m start period g))
              (else (let D ((m* m) (i 0) (acc '()))
                      (cond ((null? m*) (reverse acc))
                            ((list-tail/or-false (g (car m*)) i)
                             =>
                             (λ (t) (D (cdr m*) (add1 i) (cons (car t) acc))))
                            (else (reverse acc)))))))))
  (define (fail . args) '()))

test/pure: pass

A pure value repeats forever as a one-element circular list, so write never returns on it; zipping it with a finite list gives a finite result. 
(define (test/pure _)
  (let ((l (do/monad (← x ,3) (← y ,4) (← z ,5) ,(+ x y z))))
    (⊨ (eq? l (cdr l)))
    (⊦= 12 (car l))
    (⊦= '(12 13 14) (do/monad (← x l) (← y '(0 1 2)) ,(+ x y)))))

test/circular: pass

A circular list of any period zips too: the diagonal is finite as soon as an inner list of the cycle is, otherwise it is circular again, so (>>= m return) gives back m. 
(define (test/circular _)
  (⊦= '(1 2) (>>= (list/circular 1 2) (λ (x) (list x x))))
  (⊦= '((1 a) (2 b) (1 c))
      (do/monad (← x (list/circular 1 2)) (← y '(a b c)) ,(list x y)))
  (let ((r (>>= (list/circular 1 2) return)))
    (⊦= '(0 . 2) (list-rho r))
    (⊦= '(1 2) (take r 2)))
  (let ((r (do/monad
             (← x (list/circular 1 2 3))
             (← y (list/circular 'a 'b))
             ,(list x y))))
    (⊦= '(0 . 6) (list-rho r))
    (⊦= '((1 a) (2 b) (3 a) (1 b) (2 a) (3 b) (1 a)) (take r 7)))
  (let ((r (>>= (cons 0 (cons 9 (list/circular 1 2 3)))
                (λ (x) (cons x (cons x (list/circular x 'y)))))))
    (⊦= '(2 . 6) (list-rho r))
    (⊦= '(0 9 1 y 3 y 2 y 1 y) (take r 10))))

test/truncation: pass

The diagonal stops at the first inner list too short to reach it, rather than skipping it and sliding the later elements out of their index; a value that does not match the pattern of a ← clause goes to fail, the empty list, so the diagonal stops there too. 
(define (test/truncation _)
  (⊦= '(x) (>>= '(1 2 3) (λ (v) (if (= v 2) '(a) '(x y z w)))))
  (⊦= '((1 a c d) (2 b c* e))
      (do/monad
        (← x '(1 2 3))
        (← y '(a b))
        (← z '(c c*))
        (← r '(d e f g))
        ,(list x y z r)))
  (⊦= '(1) (do/monad (← (,a unquote b) '((1 . 2) 3 (4 . 5))) ,a)))

test/associativity: pass

Both sides of the associativity law agree index by index. On ragged inner lists one side may stop earlier: the inner bind on the right truncates at the first index any of its inner lists misses, even when the left side never looks at that index, which is why finite zip lists admit no lawful monad (Haskell's ZipList is only an Applicative). 
(define (test/associativity _)
  (let* ((table '#((1 2 0) (2) (0 1 2 1) (3 0 3 2 1)))
         (f (λ (v) (vector-ref table v)))
         (g (λ (v) (reverse (vector-ref table v)))))
    (for-each
      (λ (m)
         (let ((lhs (>>= (>>= m f) g)) (rhs (>>= m (λ (x) (>>= (f x) g)))))
           (⊦= (map (λ (l r) l) lhs rhs) (map (λ (l r) r) lhs rhs))))
      '((0 1 2 3) (2 0 1) (0 2 0 1) (1 0 2) (3 3 3 0) (2 2 2) (3 0)))
    (⊦= '(1 2 3) (>>= (>>= '(3 3 3) f) g))
    (⊦= '(1 2 3) (>>= '(3 3 3) (λ (x) (>>= (f x) g))))))
