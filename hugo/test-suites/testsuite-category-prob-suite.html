Introduction

This section tests the probability monad, where a value of type (Prob a) is a distribution: an association list of outcomes paired with their probabilities. Binding multiplies the probabilities along each path and consolidates outcomes that share the same value by summing their probabilities.

Implementation

(module
  (aux category prob)
  *
  (import
    scheme
    (chicken base)
    (chicken fixnum)
    (only srfi-1 append-map find)
    (only srfi-69
          make-hash-table
          equal?-hash
          hash-table-ref/default
          hash-table-set!)
    (aux base))
  (define (return v) (list (cons v 1)))
  (define (>>= m f)
    (consolidate
      (append-map
        (λ1-match/first
          ((,v unquote p)
           (map (λ1-match/first ((,v* unquote p*) (cons v* (* p p*)))) (f v))))
        m)))
  (define (fail . args) '())
  (define (coin/tf p) (list (cons #t p) (cons #f (- 1 p))))
  (define (consolidate m)
    (let loop ((m m) (acc '()) (n 0) (cells #f))
      (cond ((null? m) (reverse acc))
            ((and (not cells) (fx> n 60))
             (let1 (cells (make-hash-table equal? equal?-hash))
                   (for-each
                     (λ (cell) (hash-table-set! cells (car cell) cell))
                     acc)
                   (loop m acc n cells)))
            (else
             (let* ((vp (car m))
                    (v (car vp))
                    (p (cdr vp))
                    (cell (if cells
                            (hash-table-ref/default cells v #f)
                            (find (λ1-match/first ((,v* . _) (equal? v v*)))
                                  acc))))
               (cond (cell
                      (set-cdr! cell (+ (cdr cell) p))
                      (loop (cdr m) acc n cells))
                     (else
                      (let1 (cell (cons v p))
                            (when cells (hash-table-set! cells v cell))
                            (loop (cdr m) (cons cell acc) (fx+ n 1) cells))))))))))

test/return: pass

Return is the certain distribution, with the exact probability 1 so that the right identity law (>>= m return) = m holds for exact distributions too. 
(define (test/return _)
  (⊨ (equal? '((a . 1)) (return 'a)))
  (let ((d '((a . 1/2) (b . 1/2)))) (⊨ (equal? d (>>= d return)))))

test/exact: pass

Distributions built from exact rationals stay exact through binds, and their total probability is exactly 1. 
(define (test/exact _)
  (let ((d (do/monad (← x (coin/tf 1/3)) (← y (coin/tf 1/7)) ,(and x y))))
    (⊨ (equal? '((#t . 1/21) (#f . 20/21)) d))
    (⊨ (exact? (apply + (map cdr d))))))

test/consolidate: pass

consolidate sums the probabilities of outcomes sharing a value, in first-occurrence order and without mutating its argument; past about 60 distinct values it finds the summed cells through a hash table instead of scanning them. 
(define (test/consolidate _)
  (let* ((up (map (λ (i) (cons i 1/300)) (iota 100)))
         (down (map (λ (i) (cons (- 99 i) 1/150)) (iota 100)))
         (m (append up down))
         (m* (map (λ (vp) (cons (car vp) (cdr vp))) m)))
    (⊨ (equal?
         '((#f . 0.5) (#t . 0.5))
         (consolidate '((#f . 0.25) (#t . 0.5) (#f . 0.25)))))
    (⊨ (equal? (map (λ (i) (cons i 1/100)) (iota 100)) (consolidate m)))
    (⊨ (equal? m* m))))
