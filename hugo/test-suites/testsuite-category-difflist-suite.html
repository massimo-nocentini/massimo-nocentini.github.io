Introduction

This section tests difference lists: a difflist is a function that prepends its content to the list it receives, so appending two of them is function composition, in constant time, and difflist->list applies the result to '().

Implementation

(module
  (aux category difflist)
  *
  (import
    scheme
    (chicken base)
    (chicken pretty-print)
    (only srfi-1 append-map append-reverse)
    (aux base))
  (define-record category-difflist f)
  (set-record-printer!
    category-difflist
    (λ (d port)
       (let ((lst (difflist->list d)) (D (display/port port)))
         (D "⸨")
         (for-each (μ v (D " ") (D v)) lst)
         (D " ⸩"))))
  (define (return x) (make-category-difflist (λ (x*) (cons x x*))))
  (define (>>= m g)
    (let* ((f (category-difflist-f m)) (xs (map g (f '()))))
      (make-category-difflist
        (λ (xs*) (foldr (λ (m* lst) ((category-difflist-f m*) lst)) xs* xs)))))
  (define (fail . args) mempty)
  (define mempty (make-category-difflist (λ (lst) lst)))
  (define (mappend l1 l2)
    (make-category-difflist
      (o (category-difflist-f l1) (category-difflist-f l2))))
  (define (mlog x) (make-category-difflist (λ (xs) (cons x xs))))
  (define (list->difflist l1) (make-category-difflist (λ (l2) (append l1 l2))))
  (define (difflist->list dl) ((category-difflist-f dl) '()))
  (define (reverse/difflist dl)
    (make-category-difflist
      (λ (xs) (append-reverse ((category-difflist-f dl) '()) xs)))))

test/monoid: pass

Appending is composition, and mempty is the identity. 
(define (test/monoid _)
  (⊦= '(1 2 3 4 5)
      (difflist->list
        (mappend (list->difflist '(1 2 3)) (list->difflist '(4 5)))))
  (⊦= '(1 2) (difflist->list (mappend mempty (list->difflist '(1 2)))))
  (⊦= '(1 2) (difflist->list (mappend (list->difflist '(1 2)) mempty))))

test/fail: pass

A failure is the empty difflist, so it drops only its own branch, as '() does in the list monad. 
(define (test/fail _)
  (⊦= '() (difflist->list (fail)))
  (⊦= '(1 2) (difflist->list (mappend (fail) (list->difflist '(1 2)))))
  (⊦= '(1 3)
      (difflist->list
        (>>= (list->difflist '(1 2 3))
             (λ (x) (if (= x 2) (fail) (return x))))))
  (⊦= '(1 3 9 9)
      (difflist->list
        (mappend
          (>>= (list->difflist '(1 2 3))
               (λ (x) (if (= x 2) (fail) (return x))))
          (list->difflist '(9 9))))))

test/reverse: pass

Reversing a difflist reverses its own content only, not the tail it is later given. 
(define (test/reverse _)
  (⊦= '(3 2 1) (difflist->list (reverse/difflist (list->difflist '(1 2 3)))))
  (⊦= '(3 2 1 4 5)
      (difflist->list
        (mappend
          (reverse/difflist (list->difflist '(1 2 3)))
          (list->difflist '(4 5)))))
  (⊦= '(4 5 3 2 1)
      (difflist->list
        (mappend
          (list->difflist '(4 5))
          (reverse/difflist (list->difflist '(1 2 3)))))))

test/mlog: pass

Logging a value appends it as it is, as the list monoid does, so a writer logs the same over both. 
(define (test/mlog _)
  (⊦= '(a) (difflist->list (mlog 'a)))
  (⊦= '(1 2)
      (difflist->list
        (W:category-writer-log (W:>>= (W:tell 1) (λ (_) (W:tell 2)))))))

test/printer: pass

A difflist prints its content, left to right.
