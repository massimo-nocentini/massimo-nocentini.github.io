(define (test-λ-calculus-interpreter _)
  (define (lookup x env)
    (match/non-overlapping
      env
      (() (error 'lookup "unbound variable" x))
      ((((,y unquote v) unquote rest) ⊣ (equal? x y)) v)
      ((((,y unquote v) unquote rest) ⊣ (not (equal? x y)))
       (lookup x rest))))
  (define (not-in-env? x env)
    (match/non-overlapping
      env
      (() #t)
      ((((,y unquote v) unquote rest) ⊣ (equal? x y)) #f)
      ((((,y unquote v) unquote rest) ⊣ (not (equal? x y)))
       (not-in-env? x rest))))
  (define rator?
    (let1 (op-names '(lambda quote list))
          (λ (x env)
              (not (and (symbol? x) (memq x op-names) (not-in-env? x env))))))
  (define (eval-exp exp env)
    (match/non-overlapping
      exp
      (((,rator ,rand) ⊣ (rator? rator env))
       (let ((proc (eval-exp rator env)) (arg (eval-exp rand env)))
         (match/non-overlapping
           proc
           ((closure ,x ,body ,env2)
            (eval-exp body `((,x unquote arg) unquote env2))))))
      (((λ (,x) ,body) ⊣ (and (symbol? x) (not-in-env? 'λ env)))
       `(closure ,x ,body ,env))
      ((',v ⊣ (not-in-env? 'quote env)) v)
      (((list unquote a∗) ⊣ (not-in-env? 'list env))
       (map (λ (e) (eval-exp e env)) a∗))
      ((,x ⊣ (symbol? x)) (lookup x env))))
  (define-syntax-rule
    (eval/env0 body)
    (eval-exp 'body (interaction-environment/symbols '())))
  (⊦ equal?
       `(closure z z ())
       (eval/env0 (((λ (x) (λ (y) x)) (λ (z) z)) (λ (a) a))))

       `(closure y x ((x closure z z ())))
       (eval/env0 ((λ (x) (λ (y) x)) (λ (z) z))))
  (⊦ equal? `(closure y y ()) (eval/env0 ((λ (x) x) (λ (y) y))))
  (⊦ equal? `(closure y y ()) (eval/env0 ('(closure x x ()) (λ (y) y)))))
