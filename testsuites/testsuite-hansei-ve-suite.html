(define (test/exponential _)
  (define result
    (probcc-reify/exact
      (let loop ((p 'p) (n 10))
        (cond ((equal? 1 n) (probcc-coin p))
              (else
               (let1 (r (loop p (sub1 n)))
                     (not (equal? (probcc-coin p) r))))))))
