(module
  (aux schemer seasoned)
  *
  (import
    scheme
    (chicken base)
    (aux base)
    (aux continuation)
    (aux continuation classic))
  (define multi-insert*&co
    (λ (new old_l old_r sexp coll)
       (let M ((sexp sexp) (coll coll))
         (cond ((null? sexp) (coll '() 0 0))
               ((atom? (car sexp))
                (cond ((equal? (car sexp) old_l)
                       (M (cdr sexp)
                          (λ (new_sexp L R)
                             (coll (cons new (cons old_l new_sexp))
                                   (add1 L)
                                   R))))
                      ((equal? (car sexp) old_r)
                       (M (cdr sexp)
                          (λ (new_sexp L R)
                             (coll (cons old_r (cons new new_sexp))
                                   L
                                   (add1 R)))))
                      (else
                       (M (cdr sexp)
                          (λ (new_sexp L R)
                             (coll (cons (car sexp) new_sexp) L R))))))
               (else
                (M (car sexp)
                   (λ (new_sexp_car L_car R_car)
                      (M (cdr sexp)
                         (λ (new_sexp_cdr L_cdr R_cdr)
                            (coll (cons new_sexp_car new_sexp_cdr)
                                  (+ L_car L_cdr)
                                  (+ R_car R_cdr)))))))))))
  (define two-in-a-row?
    (λ (lat)
       (cond ((null? lat) #f)
             (else
              (let T ((preceding (car lat)) (rest (cdr lat)))
                (cond ((null? rest) #f)
                      (else
                       (or (equal? preceding (car rest))
                           (T (car rest) (cdr rest))))))))))
  (define two-in-a-row?&hop
    (λ (lat)
       (cond ((null? lat) #f)
             (else
              (letcc hop
                     (let T ((preceding (car lat)) (rest (cdr lat)))
                       (cond ((null? rest) (hop #f))
                             ((equal? preceding (car rest)) (hop #t))
                             (else (T (car rest) (cdr rest))))))))))
  (define intersect/letrec
    (λ (this that)
       (letrec ((I (λ (set)
                      (cond ((null? set) '())
                            ((member (car set) that)
                             (cons (car set) (I (cdr set))))
                            (else (I (cdr set)))))))
         (I this))))
  (define intersect
    (λ (this that)
       (letrec ((I (λ (l)
                      (match/first
                        l
                        (() '())
                        ((,i unquote is)
                         (cond ((member i that) (cons i (I is)))
                               (else (I is))))))))
         (I this))))
  (define intersect+all
    (λ (sets)
       (letcc hop
              (letrec ((intersect
                         (λ (this that)
                            (cond ((null? that) (hop '()))
                                  (else
                                   (letrec ((I (λ (l)
                                                  (match/first
                                                    l
                                                    (() '())
                                                    ((,i unquote is)
                                                     (if (member i that) (cons i (I is)) (I is)))))))
                                     (I this))))))
                       (all-null?
                         (λ (l)
                            (cond ((null? l) #t)
                                  ((null? (car l)) (all-null? (cdr l)))
                                  (else #f))))
                       (A (λ (sets)
                             (cond ((all-null? sets) (hop '()))
                                   ((null? (cdr sets)) (car sets))
                                   (else (intersect (car sets) (A (cdr sets))))))))
                (cond ((null? sets) '()) (else (A sets)))))))
  (define comb-upto-last
    (λ (atom k lat)
       (reverse
         (letcc skip
                (letrec ((R (λ (prefix k lat)
                               (match/first
                                 lat
                                 (() prefix)
                                 ((,a unquote as)
                                  (cond ((equal? atom a)
                                         (cond ((zero? k) prefix)
                                               (else (skip (R '() (sub1 k) as)))))
                                        (else (R (cons a prefix) k as))))))))
                  (R '() (sub1 k) lat))))))
  (define atom/book? (λ (x) (not (or (pair? x) (null? x)))))
  (define leftmost/awkward
    (λ (l)
       (letrec ((L (λ (ll)
                      (match/first
                        ll
                        (() '())
                        ((,first unquote rest)
                         (cond ((atom/book? first) first)
                               (else
                                (let1 (a (L first))
                                      (cond ((atom/book? a) a) (else (L rest)))))))))))
         (let1 (atom (L l)) (cond ((atom/book? atom) atom) (else l))))))
  (define leftmost/awkward+letcc
    (λ (l)
       (letrec ((L (λ (ll hop)
                      (match/first
                        ll
                        (() '())
                        ((,first unquote rest)
                         (cond ((atom/book? first) (hop first))
                               (else
                                (let1 (a (L first hop))
                                      (cond ((atom/book? a) a) (else (L rest hop)))))))))))
         (let1 (atom (letcc hop (L l hop)))
               (cond ((atom/book? atom) atom) (else l))))))
  (define leftmost/escape+explicit
    (λ (l)
       (letrec ((L (λ (ll out)
                      (cond ((null? ll) '(no symbol here))
                            ((atom/book? (car ll)) (out (car ll)))
                            (else (begin (L (car ll) out) (L (cdr ll) out)))))))
         (escapecc (hop (L l hop)) (else l)))))
  (define leftmost/escape
    (λ (l)
       (escapecc
         (hop (let L ((ll l))
                (match/first
                  ll
                  (() '(no symbol here))
                  ((,first unquote rest)
                   (cond ((atom/book? first) (hop first))
                         (else (begin (L first) (L rest))))))))
         (else l))))
  (define rember1*/letcc
    (λ (atom sexp)
       (letrec ((R (λ (sexp skip)
                      (match/first
                        sexp
                        (() (skip 'no-present))
                        ((,a unquote d)
                         (cond ((atom/book? a)
                                (cond ((equal? a atom) d)
                                      (else (cons a (R d skip)))))
                               (else
                                (let1 (new-car (apply/cc R (list a)))
                                      (cond ((symbol? new-car) (cons a (R d skip)))
                                            (else (cons new-car d)))))))))))
         (let1 (s (apply/cc R (list sexp)))
               (cond ((symbol? s) sexp) (else s))))))
  (define rember1*/try
    (λ (atom sexp)
       (letrec ((make-skipper
                  (λ (skip sexp)
                     (τ (let1 (reason `(,atom not present in ,sexp))
                              (display reason)
                              (skip reason)))))
                (R (λ (sexp keep-searching)
                      (match/first
                        sexp
                        (() (keep-searching))
                        ((,a unquote d)
                         (cond ((atom/book? a)
                                (cond ((equal? a atom) d)
                                      (else (cons a (R d keep-searching)))))
                               (else
                                (trycc/named
                                  (skip (cons (R a (make-skipper skip a)) d))
                                  (else (cons a (R d keep-searching)))))))))))
         (trycc/named
           (skip (R sexp (make-skipper skip sexp)))
           (else => (λ (previous) (display previous) sexp)))))))

(define (test/multi-insert*&co _)
  (let ((sexp '((((orange fish)) apple anchovy) apple (orange)))
        (sexp-expected
          '((((orange tuna fish)) apple tuna anchovy) apple (orange tuna)))
        (identity-abridged (λ args args)))
    (⊦= `(,sexp-expected 1 2)
        (multi-insert*&co 'tuna 'anchovy 'orange sexp identity-abridged))
    (⊦= `(,sexp-expected 1 2)
        (letcc hop
               (multi-insert*&co
                 'tuna
                 'anchovy
                 'orange
                 sexp
                 (λ args (hop args)))))))
