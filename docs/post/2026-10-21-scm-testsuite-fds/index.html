(define (test-find-and-union _)
  (unionfind-new
    (λ (U ↑ ↑! ≡ →)
       (⊦= 'absent (↑ 'absent))
       (⊦= '(a b) (↑ '(a b)))
       (⊦= 0 (length (→)))
       (≡ 3 4)
       (⊦= 4 (↑ 3))
       (⊦= 4 (↑ 4))
       (let1 (edges (→))
             (⊦= '(3 . 4) (assoc 3 edges))
             (⊦= '(4 . 4) (assoc 4 edges)))
       (let1 (s (with-output-to-string (λ () (display U))))
             (⊨ (and (substring-index "(4 -> 4)\n" s) #t))
             (⊨ (and (substring-index "(3 -> 4)\n" s) #t))

(define (test-edges-size-copy _)
  (let* ((U (unionfind-empty)) (≡ (unionfind-≡ U)))
    (⊨ (unionfind? U))
    (⊦= 0 (unionfind-size U))
    (⊦= 0 (unionfind-edges U))
    (≡ 1 2)
    (≡ 3 4)
    (⊦= 4 (unionfind-size U))
    (⊦= 2 (unionfind-edges U))
    (⊦= '(1 2 3 4) (sort (unionfind-keys U) <))
    (let* ((W (unionfind-copy U)) (≡W (unionfind-≡ W)))
      (≡W 2 4)
      (≡W 5 6)
      (⊦= 6 (unionfind-size W))
      (⊦= 4 (unionfind-edges W))
      (⊦= ((unionfind-↑ W) 1) ((unionfind-↑ W) 3))
      (⊦= 4 (unionfind-size U))
      (⊦= 2 (unionfind-edges U))
      (⊦= 2 ((unionfind-↑ U) 1))
      (⊦= 4 ((unionfind-↑ U) 3))
