(define (test/procc/λ-memo _)
  (define count 0)
  (define fib
    (λ (n)
       (add1! count)
       (cond ((< n 2) n)
             (else (let1 (m (sub1 n)) (+ (fib m) (fib (sub1 m))))))))
  (let1 (v (fib 10)) (⊦= 55 v) (⊦= 177 count))
  (set! count 0)
  (define fib-memo
    (λ-memo
      (n)
      (add1! count)
      (cond ((< n 2) n)
            (else (let1 (m (sub1 n)) (+ (fib-memo m) (fib-memo (sub1 m))))))))
  (let1 (v (fib-memo 10)) (⊦= 55 v) (⊦= 11 count)))

test/simdjson/frees-the-tree: pass

The C side mallocs the whole tree it hands over, and simdjson->scheme/free releases it even when the conversion raises, as it does on the error node of a rejected document: leaking that node, 200000 rejections of {bad grow the process by some 40 MB. Leaking the tree, 40 loads of twitter.json grow it by some 75 MB. An integer too large for 64 bits is refused by the C side, which frees the half-built tree and hands over such an error node, so the error carries simdjson's message, BIGINT_ERROR, and no pointer into the freed tree.

(define (test/simdjson/frees-the-tree _)
  (⊨ (eq? simdjson-load/ondemand simdjson-load))
  (⊨ (< (resident-growth (λ () (simdjson-load "twitter.json"))) 16384))
  (let1 (s (string-append
             "{\"s\":\""
             (make-string 1000000 #\x)
             "\",\"n\":123456789012345678901234567890}"))
        (⊨ (< (resident-growth
                (λ () (condition-case (simdjson-parse/ondemand s) ((exn) #f))))
              16384)))
  (let1 (rejects
          (λ ()
             (let loop ((n 5000))
               (unless
                 (zero? n)
                 (condition-case (simdjson-parse/ondemand "{bad") ((exn) #f))
                 (loop (sub1 n))))))
        (⊨ (< (resident-growth rejects) 16384)))
  (⊦= '("BIGINT_ERROR" "BIGINT_ERROR" "BIGINT_ERROR" "BIGINT_ERROR")

                 "-9223372036854775809"
                 "[123456789012345678901234567890]"
                 "{\"a\":-123456789012345678901234567890}")))
  (⊦= '()
      (condition-case
        (simdjson-parse/ondemand "[123456789012345678901234567890]")

test/simdjson/errors: pass

A document simdjson rejects raises a Scheme error carrying simdjson's own message, instead of letting the C++ exception reach std::terminate. A value starting with a character no JSON value starts with, as x or ], is an improper structure too, at the root as inside a container. On a broken structure, the children simdjson counted ahead of an array or an object may be fewer than those the walk then meets; they are checked, not written past the end of the heap block, which glibc found corrupted at a later free. The walk recurses once per nesting level, so a document nesting more than 1024 arrays or objects raises DEPTH_ERROR instead of overflowing the C stack.

(define (test/simdjson/errors _)
  (⊦= "INCOMPLETE_ARRAY_OR_OBJECT"
      (simdjson-error-code (λ () (simdjson-parse/ondemand "{bad json"))))
  (⊦= "INCOMPLETE_ARRAY_OR_OBJECT"
      (simdjson-error-code (λ () (simdjson-parse/ondemand "[1,2,{\"a\":[1,"))))
  (⊦= "NUMBER_ERROR"
      (simdjson-error-code (λ () (simdjson-parse/ondemand "[1, 2, 1e999]"))))
  (⊦= "IO_ERROR"
      (simdjson-error-code
        (λ () (simdjson-load/ondemand "no-such-file.json"))))
  (⊦= "TAPE_ERROR"
      (simdjson-error-code
        (λ () (simdjson-parse/ondemand "[[}],1,2,3,4,5,6,7,8]"))))
  (⊦= "TAPE_ERROR"
      (simdjson-error-code
        (λ () (simdjson-parse/ondemand "{\"\":[[}],\"\":\"\"}"))))
  (⊦= '("TAPE_ERROR"
        "TAPE_ERROR"
        "TAPE_ERROR"
        "TAPE_ERROR"
        "TAPE_ERROR"
        "TAPE_ERROR")
      (map (λ (s) (simdjson-error-code (λ () (simdjson-parse/ondemand s))))
