(aux category list): the list monad

Introduction

This section tests the list monad, which is a monad that represents non-deterministic computations. A value of type (list a) represents a non-deterministic computation that can produce any of the values in the list. The monad operations allow us to combine these computations in various ways, such as sequencing them with >>= or combining them with ⊕/monad.

Implementation

The list monad is implemented as follows, which defines the monad operations for lists. The return operation creates a singleton list, and the >>= operation uses append-map to apply a function to each element of the list and concatenate the results. The ⊕/monad operation is defined as append, which combines two lists of computations.

(module
  (aux category list)
  *
  (import scheme (chicken base) (only srfi-1 append-map) (aux base))
  (define (return v) (list v))
  (define (>>= m f) (append-map f m))
  (define (fail . args) '())
  (define mempty '())
  (define (mappend l1 l2) (append l1 l2))
  (define (mlog x) (list x))
  (define map/functor map)
  (define (map/applicative flst lst) (append-map (λ (f) (map f lst)) flst))
  (define ⊕₀/monad '())
  (define (⊕/monad m1 m2) (append m1 m2)))

(aux category maybe): the maybe monad

Introduction

This section tests the maybe monad, where (just . v) is a computation that succeeded with the value v and the empty list is a computation that failed, short-circuiting every later bind.

Implementation

(module
  (aux category maybe)
  *
  (import scheme (chicken base) (aux base))
  (define (return v) `(just unquote v))
  (define (>>= m f) (match/first m ((just unquote v) (f v)) (else '())))
  (define (fail . args) '()))

(aux category prob): the probability monad

Introduction

This section tests the probability monad, where a value of type (Prob a) is a distribution: an association list of outcomes paired with their probabilities. Binding multiplies the probabilities along each path and consolidates outcomes that share the same value by summing their probabilities.

Implementation

(module
  (aux category prob)
  *
  (import
    scheme
    (chicken base)
    (chicken fixnum)
    (only srfi-1 append-map find)
    (only srfi-69
          make-hash-table
          equal?-hash
          hash-table-ref/default
          hash-table-set!)
    (aux base))
  (define (return v) (list (cons v 1)))
  (define (>>= m f)
    (consolidate
      (append-map
        (λ1-match/first
          ((,v unquote p)
           (map (λ1-match/first ((,v* unquote p*) (cons v* (* p p*)))) (f v))))
        m)))
  (define (fail . args) '())
  (define (coin/tf p) (list (cons #t p) (cons #f (- 1 p))))
  (define (consolidate m)
    (let loop ((m m) (acc '()) (n 0) (cells #f))
      (cond ((null? m) (reverse acc))
            ((and (not cells) (fx> n 60))
             (let1 (cells (make-hash-table equal? equal?-hash))
                   (for-each
                     (λ (cell) (hash-table-set! cells (car cell) cell))
                     acc)
                   (loop m acc n cells)))
            (else
             (let* ((vp (car m))
                    (v (car vp))
                    (p (cdr vp))
                    (cell (if cells
                            (hash-table-ref/default cells v #f)
                            (find (λ1-match/first ((,v* . _) (equal? v v*)))
                                  acc))))
               (cond (cell
                      (set-cdr! cell (+ (cdr cell) p))
                      (loop (cdr m) acc n cells))
                     (else
                      (let1 (cell (cons v p))
                            (when cells (hash-table-set! cells v cell))
                            (loop (cdr m) (cons cell acc) (fx+ n 1) cells))))))))))

(aux category ziplist): the ziplist monad

Introduction

This section tests the ziplist monad, whose bind zips rather than nests: the i-th element of (>>= m g) is the i-th element of (g v), where v is the i-th element of m. A pure value is the circular list that repeats it, so it zips with lists of any length.

Implementation

(module
  (aux category ziplist)
  *
  (import scheme (chicken base) (only srfi-1 take every iota) (aux base))
  (define (list-tail/or-false lst i)
    (cond ((null? lst) #f)
          ((zero? i) lst)
          (else (list-tail/or-false (cdr lst) (sub1 i)))))
  (define (list-rho lst)
    (let meet ((t lst) (h lst))
      (cond ((or (null? h) (null? (cdr h))) #f)
            ((eq? (cdr t) (cddr h))
             (let* ((t (cdr t))
                    (start (let S ((a lst) (b t) (n 0))
                             (if (eq? a b) n (S (cdr a) (cdr b) (add1 n)))))
                    (period
                      (let P ((b (cdr t)) (n 1))
                        (if (eq? b t) n (P (cdr b) (add1 n))))))
               (cons start period)))
            (else (meet (cdr t) (cddr h))))))
  (define (>>=/circular m start period g)
    (let* ((images (list->vector (map g (take m (+ start period)))))
           (at (λ (i)
                  (vector-ref
                    images
                    (if (< i start) i (+ start (modulo (- i start) period))))))
           (rhos (map (λ (k) (list-rho (vector-ref images k)))
                      (iota period start)))
           (from (and (every pair? rhos) (apply max start (map car rhos))))
           (cycle (and from (apply lcm period (map cdr rhos))))
           (bound (and from (+ from cycle)))
           (d (let D ((i 0) (acc '()))
                (cond ((eqv? i bound) (reverse acc))
                      ((list-tail/or-false (at i) i)
                       =>
                       (λ (t) (D (add1 i) (cons (car t) acc))))
                      (else (reverse acc))))))
      (when (and bound (= (length d) bound))
            (set-cdr! (list-tail d (sub1 bound)) (list-tail d from)))
      d))
  (define (return x) (list/circular x))
  (define (>>= m g)
    (match/first
      m
      ((,l unquote ,m) (g l))
      (else (match/first
              (list-rho m)
              ((,start unquote period) (>>=/circular m start period g))
              (else (let D ((m* m) (i 0) (acc '()))
                      (cond ((null? m*) (reverse acc))
                            ((list-tail/or-false (g (car m*)) i)
                             =>
                             (λ (t) (D (cdr m*) (add1 i) (cons (car t) acc))))
                            (else (reverse acc)))))))))
  (define (fail . args) '()))

(aux category difflist): difference lists

Introduction

This section tests difference lists: a difflist is a function that prepends its content to the list it receives, so appending two of them is function composition, in constant time, and difflist->list applies the result to '().

Implementation

(module
  (aux category difflist)
  *
  (import
    scheme
    (chicken base)
    (chicken pretty-print)
    (only srfi-1 append-map append-reverse)
    (aux base))
  (define-record category-difflist f)
  (set-record-printer!
    category-difflist
    (λ (d port)
       (let ((lst (difflist->list d)) (D (display/port port)))
         (D "⸨")
         (for-each (μ v (D " ") (D v)) lst)
         (D " ⸩"))))
  (define (return x) (make-category-difflist (λ (x*) (cons x x*))))
  (define (>>= m g)
    (let* ((f (category-difflist-f m)) (xs (map g (f '()))))
      (make-category-difflist
        (λ (xs*) (foldr (λ (m* lst) ((category-difflist-f m*) lst)) xs* xs)))))
  (define (fail . args) mempty)
  (define mempty (make-category-difflist (λ (lst) lst)))
  (define (mappend l1 l2)
    (make-category-difflist
      (o (category-difflist-f l1) (category-difflist-f l2))))
  (define (mlog x) (make-category-difflist (λ (xs) (cons x xs))))
  (define (list->difflist l1) (make-category-difflist (λ (l2) (append l1 l2))))
  (define (difflist->list dl) ((category-difflist-f dl) '()))
  (define (reverse/difflist dl)
    (make-category-difflist

(aux category state): the state monad

Introduction

This section tests the state monad, Haskell's State s a, where a computation is a transition from a state to the pair of its result and the next state, Haskell's (a, s) tuple; binding threads the state from one computation to the next, strictly, as Haskell's Control.Monad.State.Strict does.

Implementation

A computation is a category-state record around its transition, the image of Haskell's newtype, so that a computation is a type of its own and not any procedure; run-state applies the transition to a state.

(module
  (aux category state)
  *
  (import scheme (chicken base) (aux base))
  (define-record category-state run)
  (set-record-printer! category-state (λ (m port) (display "#<state>" port)))
  (define (run-state m s) ((category-state-run m) s))
  (define (eval-state m s) (car (run-state m s)))
  (define (exec-state m s) (cdr (run-state m s)))
  (define (return v) (make-category-state (λ (s) (cons v s))))
  (define (>>= m f)
    (make-category-state
      (λ (s) (let1 (r (run-state m s)) (run-state (f (car r)) (cdr r))))))
  (define (fail . args)
    (make-category-state
      (λ (s)
         (error "State monad does not support failure"
                `((on ,s) (args unquote args))))))
  (define get (make-category-state (λ (s) (cons s s))))
  (define (put s) (make-category-state (λ (_) (cons (void) s))))
  (define (modify f) (make-category-state (λ (s) (cons (void) (f s)))))
  (define (gets f) (make-category-state (λ (s) (cons (f s) s))))
  (define (state f) (make-category-state f))
  (define (with-state f m) (make-category-state (λ (s) (run-state m (f s))))))
