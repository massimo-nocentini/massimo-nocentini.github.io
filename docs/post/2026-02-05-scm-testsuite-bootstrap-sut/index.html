The (aux unittest) module

Bootstrapping a unittest framework

 Posted on February 5, 2026  (Last modified on February 11, 2026)  |   8 min  |   Massimo Nocentini

This test suite drove the implementation of the unittest framework itself.

(module
  (aux unittest)
  *
  (import
    scheme
    (scheme case-lambda)
    (chicken base)
    (chicken condition)
    (chicken pretty-print)
    (chicken port)
    (chicken string)
    (chicken syntax)
    (chicken flonum)
    (chicken time)
    (chicken gc)
    srfi-1
    srfi-19
    (aux sxml))
  (define-record unittest/testcase name log)
  (define (unittest/wasrun name) (make-unittest/testcase name '()))
  (define (unittest/testcase-logcons! testcase msg)
    (unittest/testcase-log-set!
      testcase
      (cons msg (unittest/testcase-log testcase))))
  (define (unittest/testcase-guard result testcase-name witness thunk)
    (condition-case
      (thunk)
      (c (exn unittest-assert-equal)
         (unittest/result-failed!
           result
           (cons testcase-name
                 (get-condition-property
                   c
                   'unittest-assert-equal
                   'comparison)))
         witness)
      (c (exn)
         (unittest/result-failed!
           result
           (list testcase-name
                 (call-with-output-string
                   (lambda (port) (print-error-message c port)))))
         witness)
      (c () (unittest/result-failed! result (list testcase-name c)) witness)))
  (define (unittest/testcase-run testcase result sut)
    (let* ((methods (cdr sut))
           (setup (alist-ref 'setup methods))
           (teardown (alist-ref 'teardown methods))
           (testcase-name (unittest/testcase-name testcase))
           (witness (gensym))
           (guarded
             (lambda (thunk)
               (unittest/testcase-guard result testcase-name witness thunk))))
      (unittest/result-started! result)
      (let ((args (if setup
                    (guarded
                      (lambda ()
                        (call-with-values
                          (lambda () ((car setup) testcase))
                          list)))
                    '())))
        (let-values
          (((f code) (apply values (alist-ref testcase-name methods))))
          (let* ((v witness)
                 (eta-time 0)
                 (outstr "")
                 (errstr
                   (if (eq? args witness)
                     ""
                     (with-error-output-to-string
                       (lambda ()
                         (set! outstr
                           (with-output-to-string
                             (lambda ()
                               (let-values
                                 (((u s) (cpu-time)))
                                 (set! v
                                   (guarded (lambda () (apply f testcase args))))
                                 (let-values
                                   (((uu ss) (cpu-time)))
                                   (set! eta-time (- (+ uu ss) (+ u s))))))))))))
                 (hasdoc (and (pair? v) (eq? (car v) 'doc)))
                 (torn (cond ((or (not teardown) (eq? args witness)) (void))
                             ((eq? v witness)
                              (condition-case
                                (apply (car teardown) testcase args)
                                (c ()
                                   (set! errstr
                                     (string-append
                                       errstr
                                       (call-with-output-string
                                         (lambda (port) (print-error-message c port)))))
                                   witness)))
                             (else
                              (guarded
                                (lambda () (apply (car teardown) testcase args)))))))
            (let ((header-id (symbol->string testcase-name)))
              `((h2 (@ (id ,header-id))
                    (a (@ (href ,(string-append "#" header-id)))
                       (code ,header-id)
                       ": "
                       ,(if (or (eq? v witness) (eq? torn witness))
                          '(span (@ (class "w3-text-red")) fail)
                          '(span (@ (class "w3-text-green")) pass))))
                ,@(if hasdoc (cdr v) '())
                (code/scheme ,(if hasdoc (butlast code) code))
                (code/scheme
                  ((eta ,(exact->inexact (/ eta-time 1000)))
                   (memory ,(memory-statistics))
                   (stdout ,outstr)
                   (stderr ,errstr))))))))))
  (define (unittest/suite sut)
    (let loop ((names (map car (cdr sut))))
      (cond ((null? names) sut)
            ((memq (car names) (cdr names))
             (error 'define-suite
                    "duplicate test case name"
                    (car sut)
                    (car names)))
            (else (loop (cdr names))))))
  (define-syntax
    define-suite
    (syntax-rules
      ()
      ((_ sutname ((casename formal ...) body ...) ...)
       (define sutname
         (unittest/suite
           `(sutname
              (casename
                ,(lambda (formal ...) body ...)
                ,'(define (casename formal ...) body ...))
              ...))))))
  (define-syntax
    lettest
    (syntax-rules
      ()
      ((_ ((test nameexp) ...) body ...)
       (let ((test (unittest/wasrun nameexp)) ...) body ...))))
  (define-record unittest/result ran failed)
  (define (unittest/result-summary result)
    `((ran ,(unittest/result-ran result))
      (failed
        ,(length (unittest/result-failed result))
        ,@(unittest/result-failed result))))
  (define (unittest/result-started! result)
    (unittest/result-ran-set! result (add1 (unittest/result-ran result))))
  (define (unittest/result-failed! result exn)
    (unittest/result-failed-set!
      result
      (cons exn (unittest/result-failed result))))
  (define (equal-approx? p)
    (letrec ((? (lambda (a b)
                  (cond ((and (real? a) (real? b))
                         (or (= a b) (< (abs (- a b)) p)))
                        ((and (pair? a) (pair? b))
                         (and (? (car a) (car b)) (? (cdr a) (cdr b))))
                        ((and (vector? a) (vector? b))
                         (and (= (vector-length a) (vector-length b))
                              (every ? (vector->list a) (vector->list b))))
                        (else (equal=? a b))))))
      ?))
  (define (⊦ pred? . args)
    (unless
      (apply pred? args)
      (signal (apply unittest/condition-expected-actual args))))
  (define (⊦= a b) (⊦ (equal-approx? 1e-06) a b))
  (define (⊦≠ a b) (⊦ (complement equal?) a b))
  (define (⊨ a) (⊦= #t a))
  (define (⊭ a) (⊦= #f a))
  (define-syntax
    letsuite
    (syntax-rules
      ()
      ((_ ((name '(method ...)) ...) body ...)
       (letrec ((name (lettest ((method 'method) ...) (list method ...))) ...)
         body
         ...))))
  (define (unittest/testsuite-run suite r sut)
    (map (lambda (testcase) (unittest/testcase-run testcase r sut)) suite))
  (define (unittest/✓ sut)
    (let* ((r (make-unittest/result 0 '()))
           (sut-name (car sut))
           (sut-methods (cdr sut))
           (F (lambda (x)
                (let ((name (car x)))
                  (and (not (eq? name 'setup))
                       (not (eq? name 'teardown))
                       (not (eq? name 'doc))))))
           (methods (filter F sut-methods))
           (s (map (lambda (pair) (lettest ((t (car pair))) t)) methods)))
      (let ((docs (unittest/testsuite-run s r sut))
            (res (unittest/result-summary r))
            (sxml (alist-ref 'doc sut-methods)))
        (SXML->HTML->file!
          (sxml-tree sut-name `(,@(if sxml ((car sxml) r) '()) ,@docs))
          (symbol-append 'testsuite- sut-name))
        (pretty-print res)
        (unless
          (null? (unittest/result-failed r))
          (implicit-exit-handler (lambda () (exit 1))))
        r)))
  (define unittest/condition-expected-actual
    (case-lambda
      ((a b)
       (condition
         `(exn message "assert-equal failed")
         `(unittest-assert-equal comparison ((expected ,a) (got ,b)))))
      (args (condition
              `(exn message "assert failed")
              `(unittest-assert-equal comparison ((got ,@args)))))))
  (define-syntax
    ⊦⧳
    (syntax-rules
      ()
      ((_ ((exn ...) ...) body ...)
       (condition-case (begin body ...) ((exn ...) (void)) ...))))
  (define-syntax
    ⊦raises
    (syntax-rules
      ()
      ((_ (kind ...) body ...)
       (⊦= 'raised
           (condition-case
             (begin body ... 'not-raised)
             ((kind ...) 'raised)))))))
